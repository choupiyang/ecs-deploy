# 安装（uv 推荐）：
$ git clone https://github.com/NanmiCoder/MediaCrawler
$ cd MediaCrawler && uv sync
# 连接 Chrome（CDP 复用登录态）
$ uv run main.py --platform xhs \
  --lt qrcode --type search
# 扫二维码登录 → 自动爬取关键词笔记
